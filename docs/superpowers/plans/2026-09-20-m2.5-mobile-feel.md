```gdscript
## control_settings.gd
## NO TS COUNTERPART — authorised by M2.5 spec §4.1.
class_name ControlSettings
extends RefCounted
## Player-chosen control scheme and tilt tuning, persisted across launches.
##
## The web game has no settings surface for these: GameOptions.ts:80 fixes the
## control scheme at construction and GyroscopeController.ts exposes no
## sensitivity. Both are M2.5 divergences, authorised by spec §4.1 and §5.4.
##
# TODO(M7): fold into meta/save_store.gd alongside tilt_gyro_calibration.json
# and tilt_to_live_best.json, per M2.5 spec D7.

## Where the settings are persisted.
const SETTINGS_PATH: String = "user://peril_point_settings.json"

## Sensitivity bounds. 1.0 reproduces the cited MAX_TILT_ANGLE exactly.
const MIN_SENSITIVITY: float = 0.4
const MAX_SENSITIVITY: float = 2.5
## Smoothing bounds. 0.0 reproduces unsmoothed M2 behaviour exactly.
const MIN_SMOOTHING: float = 0.0
const MAX_SMOOTHING: float = 0.9

## Which control scheme drives the ship.
var scheme: Types.ControlScheme = Types.ControlScheme.MOUSE
## Divides MAX_TILT_ANGLE. Higher means less tilt for full deflection.
var tilt_sensitivity: float = 1.0
## Interpolation weight applied to tilt_vector each poll.
var tilt_smoothing: float = 0.0


func _init() -> void:
	scheme = Types.ControlScheme.GYRO if OS.has_feature("mobile") else Types.ControlScheme.MOUSE
	_load_settings()


## Persists the current settings to SETTINGS_PATH as JSON.
func save_settings() -> void:
	var file: FileAccess = FileAccess.open(SETTINGS_PATH, FileAccess.WRITE)
	if file == null:
		Log.warn("control settings could not be written to %s" % SETTINGS_PATH)
		return
	file.store_string(
		JSON.stringify(
			{
				"scheme": int(scheme),
				"tilt_sensitivity": tilt_sensitivity,
				"tilt_smoothing": tilt_smoothing,
			}
		)
	)
	file.close()


# Mirrors the defensive JSON handling in tilt_input.gd:229-253: an unreadable
# or wrongly-typed value warns and keeps the default rather than crashing.
func _load_settings() -> void:
	if not FileAccess.file_exists(SETTINGS_PATH):
		return
	var file: FileAccess = FileAccess.open(SETTINGS_PATH, FileAccess.READ)
	if file == null:
		return
	var raw: String = file.get_as_text()
	file.close()

	# PORT-NOTE: Variant is unavoidable for unvalidated JSON (CONVENTIONS §4).
	var parsed: Variant = JSON.parse_string(raw)
	if not (parsed is Dictionary):
		Log.warn("control settings are not a JSON object; keeping defaults")
		return
	var data: Dictionary = parsed

	var scheme_value: Variant = data.get("scheme")
	if scheme_value is float:
		var ordinal: int = int(scheme_value)
		if ordinal >= Types.ControlScheme.HYBRID and ordinal <= Types.ControlScheme.GYRO:
			scheme = ordinal as Types.ControlScheme
		else:
			Log.warn("control settings carry an unknown scheme %d; keeping default" % ordinal)

	var sensitivity_value: Variant = data.get("tilt_sensitivity")
	if sensitivity_value is float:
		tilt_sensitivity = clampf(sensitivity_value, MIN_SENSITIVITY, MAX_SENSITIVITY)

	var smoothing_value: Variant = data.get("tilt_smoothing")
	if smoothing_value is float:
		tilt_smoothing = clampf(smoothing_value, MIN_SMOOTHING, MAX_SMOOTHING)
```

Note on `is float`: `JSON.parse_string` returns every JSON number as a float, so `is float` is the correct check for an integer-valued key such as `scheme` — the same pattern `tilt_input.gd:250` uses.

```gdscript
## Reads the device sensor and applies it. Called once per physics tick by
## GameRoot while the gyro scheme is active.
func poll() -> void:
	if not is_listening:
		return
	var g: Vector3 = Input.get_gravity()
	if g == Vector3.ZERO:
		g = Input.get_accelerometer()
	if g == Vector3.ZERO:
		return
	apply_gravity(g, _screen_angle_degrees())


## Maps a gravity vector to the steering vector, relative to the calibrated
## pose. Separated from poll() so it is testable with an injected reading.
##
# DIVERGENCE(M2.5 §5.2): replaces M2's gravity → angles → synthetic beta/gamma
# → apply_orientation() chain, which applied the screen rotation twice (once
# in the decomposition, once inside apply_orientation) and therefore
# cross-coupled the two output axes. This applies it exactly once, here.
func apply_gravity(g: Vector3, screen_angle: int) -> void:
	_has_received_data = true
	var normalised: Vector3 = g.normalized()

	if _calibration_pending:
		_calibrated_gravity = normalised
		_calibration_pending = false

			screen_up = Vector3(0.0, -1.0, 0.0)

	# which is the unit _apply_angles() and MAX_TILT_ANGLE work in.
```

```gdscript
## floating_joystick.gd
## NO TS COUNTERPART — authorised by M2.5 spec §8.
class_name FloatingJoystick
extends RefCounted
## Floating on-screen joystick: the base appears wherever the finger lands,
## the knob tracks the drag, both clear on release.
##
## The web game has no joystick — TiltGame.ts:1253-1285 steers the ship toward
## the raw pointer position instead. That behaviour is preserved in
## touch_input.gd and still drives the HYBRID and desktop schemes; this is the
## alternative the TOUCH scheme selects.
##
## All positions are playfield-local: GameRoot subtracts the arena origin
## (M2.5 spec §6.1) before calling in.

## Distance from the base at which the knob reaches full deflection.
const BASE_RADIUS: float = 90.0
## Radius inside which a resting thumb produces no steering.
const DEAD_RADIUS: float = 8.0

## True while a finger is driving the stick.
var is_active: bool = false
## Where the finger landed. Meaningless while is_active is false.
var base_position: Vector2 = Vector2.ZERO
## Where the knob is drawn, clamped to BASE_RADIUS from the base.
var knob_position: Vector2 = Vector2.ZERO
## Steering direction, magnitude 0.0 to 1.0.
var vector: Vector2 = Vector2.ZERO

# Which touch index owns the stick, or Types.ABSENT when it is idle. A second
# simultaneous finger is ignored rather than stealing control, so a stray palm
# touch cannot yank the ship mid-run.
var _owner_index: int = Types.ABSENT


## Engages the stick if no finger already owns it.
func on_touch(index: int, position: Vector2) -> void:
	if _owner_index != Types.ABSENT:
		return
	_owner_index = index
	is_active = true
	base_position = position
	knob_position = position
	vector = Vector2.ZERO


## Moves the knob, if this is the owning finger.
func on_drag(index: int, position: Vector2) -> void:
	if index != _owner_index:
		return

	var offset: Vector2 = position - base_position
	var distance: float = offset.length()

	if distance <= DEAD_RADIUS:
		knob_position = position
		vector = Vector2.ZERO
		return

	if distance > BASE_RADIUS:
		offset = offset / distance * BASE_RADIUS
		distance = BASE_RADIUS

	knob_position = base_position + offset
	vector = offset / BASE_RADIUS


## Disengages the stick, if this is the owning finger.
func on_release(index: int) -> void:
	if index != _owner_index:
		return
	reset()


## Disengages unconditionally. Used when the control scheme changes mid-touch.
func reset() -> void:
	_owner_index = Types.ABSENT
	is_active = false
	base_position = Vector2.ZERO
	knob_position = Vector2.ZERO
	vector = Vector2.ZERO
```
